cmake_minimum_required(VERSION 3.20)
project(CGEndlessRunner VERSION 0.2.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenGL REQUIRED)
find_package(GLUT REQUIRED)
find_package(glm CONFIG REQUIRED)
set(GLAD_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/external/glad" CACHE PATH "GLAD 1 root containing include/glad/glad.h and src/glad.c")
if(NOT EXISTS "${GLAD_ROOT}/src/glad.c" OR NOT EXISTS "${GLAD_ROOT}/include/glad/glad.h")
  message(FATAL_ERROR "GLAD 1 missing. Set -DGLAD_ROOT=path or use scripts/setup-cloud.sh")
endif()
add_library(glad STATIC "${GLAD_ROOT}/src/glad.c")
target_include_directories(glad PUBLIC "${GLAD_ROOT}/include")
target_link_libraries(glad PUBLIC ${CMAKE_DL_LIBS})

add_library(runner_game src/Game.cpp src/HeightField.cpp src/Player.cpp)
target_include_directories(runner_game PUBLIC src)
add_executable(endless_runner src/main.cpp src/Shader.cpp src/Obstacles.cpp src/Renderer.cpp src/SmokeTest.cpp)
target_link_libraries(endless_runner PRIVATE runner_game glad GLUT::GLUT glm::glm OpenGL::GL)
target_compile_definitions(endless_runner PRIVATE NOMINMAX FREEGLUT_LIB_PRAGMAS=0)
if(WIN32)
  target_link_libraries(endless_runner PRIVATE user32)
endif()
foreach(target runner_game endless_runner)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
# Keep shaders beside the binary, including multi-config Visual Studio builds.
add_custom_target(runner_shaders ALL
  COMMAND ${CMAKE_COMMAND} -E copy_directory
          ${CMAKE_CURRENT_SOURCE_DIR}/shaders $<TARGET_FILE_DIR:endless_runner>/shaders)
add_dependencies(endless_runner runner_shaders)

include(CTest)
if(BUILD_TESTING)
  add_executable(game_tests tests/game_tests.cpp)
  target_link_libraries(game_tests PRIVATE runner_game)
  add_test(NAME game_logic COMMAND game_tests)
  add_executable(terrain_player_tests tests/terrain_player_tests.cpp)
  target_link_libraries(terrain_player_tests PRIVATE runner_game)
  add_test(NAME terrain_player_logic COMMAND terrain_player_tests)
endif()
